📋 Integration — as simple as this
<!-- 1. Load via CDN -->
<script type="module"
  src="https://cdn.jsdelivr.net/gh/HailBytes/hailbytes-password-analyzer@main/hailbytes-password-strength.js"></script>

<!-- 2. Use the element -->
<hailbytes-password-strength theme="dark"></hailbytes-password-strength>

<!-- 3. Listen for score events (optional) -->
<script>
  document.querySelector('hailbytes-password-strength')
    .addEventListener('password-score', (e) => {
      console.log(e.detail); // { score, label, entropy, checks }
    });
</script>
